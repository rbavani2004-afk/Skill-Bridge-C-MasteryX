1. Move All Zeros to the End:

#include <stdio.h>

int main()
{
    int arr[100], n;
    int i, j = 0, temp;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter the elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    /* Move all non-zero elements to the front */
    for (i = 0; i < n; i++)
    {
        if (arr[i] != 0)
        {
            temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            j++;
        }
    }

    printf("Array after moving zeros to the end:\n");

    for (i = 0; i < n; i++)
    {
        printf("%d ", arr[i]);
    }

    printf("\n");

    return 0;
}

Output:

Enter the number of elements: 5
Enter the elements:
0 1 0 3 12

Array after moving zeros to the end:
1 3 12 0 0

2. Maximum Sum of a Contiguous Subarray:

#include <stdio.h>

int main()
{
    int arr[100], n;
    int i;
    int currentSum, maxSum;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter the elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    currentSum = arr[0];
    maxSum = arr[0];

    for (i = 1; i < n; i++)
    {
        if (currentSum + arr[i] > arr[i])
            currentSum = currentSum + arr[i];
        else
            currentSum = arr[i];

        if (currentSum > maxSum)
            maxSum = currentSum;
    }

    printf("Maximum subarray sum: %d\n", maxSum);

    return 0;
}

Output:
Enter the number of elements: 9
Enter the elements:
-2 1 -3 4 -1 2 1 -5 4

Maximum subarray sum: 6


3. Intersection of Two Arrays:

#include <stdio.h>

int main()
{
    int arr1[100], arr2[100];
    int n1, n2;
    int i, j;
    int found;

    printf("Enter the size of first array: ");
    scanf("%d", &n1);

    printf("Enter the elements of first array:\n");
    for (i = 0; i < n1; i++)
    {
        scanf("%d", &arr1[i]);
    }

    printf("Enter the size of second array: ");
    scanf("%d", &n2);

    printf("Enter the elements of second array:\n");
    for (i = 0; i < n2; i++)
    {
        scanf("%d", &arr2[i]);
    }

    printf("Intersection of the arrays: ");

    for (i = 0; i < n1; i++)
    {
        found = 0;

        for (j = 0; j < n2; j++)
        {
            if (arr1[i] == arr2[j])
            {
                found = 1;
                break;
            }
        }

        if (found)
        {
            printf("%d ", arr1[i]);
        }
    }

    printf("\n");

    return 0;
}

Output:

Enter the size of first array: 5
Enter the elements of first array:
1 2 3 4 5

Enter the size of second array: 5
Enter the elements of second array:
3 4 5 6 7

Intersection of the arrays: 3 4 5
